-- Sample Unifier BI Publisher Data Model Snippet (Cost Variance & EAC Analysis) SELECT s.shell_name, cbs.cbs_code, cbs.cbs_name, NVL(cost.orig_budget, 0) AS original_budget, NVL(cost.appr_changes, 0) AS approved_changes, (NVL(cost.orig_budget, 0) + NVL(cost.appr_changes, 0)) AS revised_budget, NVL(cost.commitments, 0) AS total_commitments, NVL(cost.actual_cost, 0) AS actual_cost, (NVL(cost.orig_budget, 0) + NVL(cost.appr_changes, 0) - NVL(cost.actual_cost, 0)) AS cost_variance FROM sys_shell s JOIN sys_cost_sheet_data cost ON s.shell_id = cost.shell_id JOIN sys_cbs_code cbs ON cost.cbs_id = cbs.cbs_id WHERE s.status = 'Active' ORDER BY s.shell_name, cbs.cbs_code;
